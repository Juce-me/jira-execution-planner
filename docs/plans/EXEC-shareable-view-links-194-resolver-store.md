- Produces (all pure; `frontend/src/shareView/resolver.js`):
```
resolveShareViewScope(state, ctx) -> { ok:true, group:{ id, teamIds }, sprint:{ id, name, state }, mode, reported:string[] }
                                   | { ok:false, reason:'group_unresolved'|'sprint_unresolved', reported:string[] }
buildShareViewFieldPlan(state, scope, ctx = {}) -> { apply:{...}, reported:string[] }
resolveShareViewTeams(state, group, mode = state.mode) -> { teams:string[]|null, reported:string[] }
resolveShareViewFacets(state, options, mode = state.mode) -> { apply:{ status?, priority?, track? }, reported:string[] }
resolveShareViewFocus(state, mode) -> null | { kind:'alertTeam', anchor, id } | { kind:'epic'|'story', key }
ctx = { visibleGroups:[{ id, teamIds }]            // visibleControlGroups
        sprints:[{ id, name, state, startDate }]   // the validated Sprint catalog (availableSprints)
        boardScopeReady?:(groupId, scope) => boolean  // Task 11: engSprintSelectorState.componentReadiness (component) or allWorkReadiness (all_work) === 'ready' for that group
        currentQuarter?:'YYYYQn' }                 // optional; defaults to getCurrentQuarterLabel()
options = { status:string[], priority:string[], track:string[] }   // the DECLARED option ids of the facets at Stage C:
          // Object.fromEntries(model.facets.map((facet) => [facet.id, facet.options.map((option) => option.id)]))
```
- Call order for Tasks 9 and 11, with the resolved `scope.mode` passed on so a fall-back to Catch Up is honored: `resolveShareViewScope` once the catalogs are ready; `buildShareViewFieldPlan(state, scope, ctx)` at Stage B; `resolveShareViewTeams(state, scope.group, scope.mode)` and `resolveShareViewFacets(state, options, scope.mode)` at Stage C; `resolveShareViewFocus(state, scope.mode)` at Stage D. The notice categories are the union of the decode-time `entry.reported` and every `reported` array returned here.
- Semantics Tasks 9 and 11 rely on:
  - `scope.mode` is the requested mode, or `'catch-up'` with `'mode'` reported when the mode is unknown, Planning meets a closed sprint or Statistics a future one (state compared lowercase; Board is always available). Unresolved group and sprint report both categories, group first.
  - `apply` carries only the keys of the resolved mode. Catch Up: `find lanes initiatives sort alerts`. Planning: `find lanes layout`, plus `rows` for the table or `initiatives sort` for the list. Board: `find`, plus `scope` (absent, with `'mode'` reported, when a strict scope is requested and `ctx.boardScopeReady` is not `true`). Statistics: `stats range` (`range` is `null` for `teams`, `priority`, `burnout`). Each value is the link value or its `SHARE_VIEW_DEFAULTS` constant. `alerts` is `{ panel, expanded }` over all ten category keys: omitted or `off` leave every category expanded (`off` sets `panel` false), `none` collapses all, a list expands exactly its keys (a key the sprint state does not render is not an error).
  - A skipped key is absent from `apply`: test `Object.hasOwn(apply, key)`. A neutral Project Track is an own property holding `undefined`.
  - `resolveShareViewTeams` returns `['all']` when the link says all or omits the key, `null` (skip, never the widened `['all']`) when nothing matches or Teams do not apply to the mode, and reports `teams` whenever an id was dropped.
  - `resolveShareViewFacets` stores each atom as the option's own id, drops unmatched atoms, skips a key left empty, and reports `filters` for either; an omitted key is its default constant, not matched against the options.
  - `resolveShareViewFocus` returns the one target Stage D should reveal. `buildShareViewFieldPlan` reports `focus` when the link carried a focus target that was not chosen (wrong mode, alerts off, Planning table rows mismatch, or more than one target).

**Forbidden regressions for this task:**
- A link selects among values the recipient owns and never reads the recipient's prefs: an omitted key is the default constant.
- Validate first, then call setters: an unmatched Team list is never widened to `['all']`, and an unmatched status, priority or Project Track id never reaches a setter.
- The group match stays exact and case-sensitive, and the Sprint keeps the catalog entry's own `id`.
- The resolver stays pure: no React, DOM, storage, network or Jira write path; closed vocabularies are checked with `Set.has`, never `obj[value]`.

test('teams: the repo vectors, and what sanitizeSelectedTeamsForScope would have done instead', () => {
    const scope = { activeGroupTeamIds: GROUP.teamIds, availableTeamIds: GROUP.teamIds };
    // all wins
    assert.deepEqual(sanitizeSelectedTeamsForScope(['t1', 'all'], scope), ['all']);
    assert.deepEqual(resolveShareViewTeams(link({ teams: ['t1', 'all'] }), GROUP).teams, ['all']);
    // a partial match keeps the rest, but only the resolver reports the dropped id
    const narrow = { id: 'g', teamIds: ['t1', 't2', 't3'] };
    assert.deepEqual(sanitizeSelectedTeamsForScope(['t1', 't4'], { activeGroupTeamIds: narrow.teamIds, availableTeamIds: narrow.teamIds }), ['t1']);
    assert.deepEqual(resolveShareViewTeams(link({ teams: ['t1', 't4'] }), narrow), { teams: ['t1'], reported: ['teams'] });
    // no match: the repo widens, the resolver must not apply that
    assert.deepEqual(sanitizeSelectedTeamsForScope(['zz'], scope), ['all']);
    assert.equal(resolveShareViewTeams(link({ teams: ['zz'] }), GROUP).teams, null);
    // duplicates pass through the repo function untouched, so the resolver de-duplicates first
    assert.deepEqual(sanitizeSelectedTeamsForScope(['t1', 't1', 't2'], scope), ['t1', 't1', 't2']);
    assert.deepEqual(resolveShareViewTeams(link({ teams: ['t1', 't1', 't2'] }), GROUP), { teams: ['t1', 't2'], reported: [] });
});

// docs/plans/SUPPORT-shareable-view-links-design-194.md, "Link contract" and "Apply machine").

const FOCUS_KINDS = new Set(['epic', 'story']);

// ctx = { visibleGroups: [{ id, teamIds }], sprints: [{ id, name, state, ... }],
//         boardScopeReady?: (groupId, scope) => boolean, currentQuarter?: 'YYYYQn' }
export function resolveShareViewScope(state, ctx) {
    const group = (ctx.visibleGroups || []).find((entry) => String(entry.id) === state.group);
    const sprint = (ctx.sprints || []).find((entry) => String(entry.id) === state.sprint);
    const reported = [];
    if (!group) reported.push('group');
    if (!sprint) reported.push('sprint');
    if (!group || !sprint) {
        return { ok: false, reason: group ? 'sprint_unresolved' : 'group_unresolved', reported };
    }
    // The predicates EngModeControl disables Planning and Statistics with; Board is always available.

        && !(state.mode === 'planning' && sprintState === 'closed')
        && !(state.mode === 'statistics' && sprintState === 'future');
    if (!available) reported.push('mode');
    return {
        ok: true,
        group: { id: group.id, teamIds: normalizeTeamIds(group.teamIds) },
        sprint: { id: sprint.id, name: sprint.name, state: sprint.state },
        mode: available ? state.mode : 'catch-up',
        reported,
    };

function teamsApply(state, mode) {
    if (mode === 'catch-up' || mode === 'planning') return true;
    return mode === 'statistics' && TEAM_STATS_VIEWS.has(linkValue(state, 'stats'));
}

// `mode` is the resolved mode; it differs from state.mode after a fall-back to Catch Up.
export function resolveShareViewTeams(state, group, mode = state.mode) {
    if (!teamsApply(state, mode)) return { teams: null, reported: [] };
    const requested = unique(linkValue(state, 'teams'));
    if (requested.includes('all')) return { teams: ['all'], reported: [] };
    const inGroup = new Set(normalizeTeamIds(group.teamIds));
    const kept = requested.filter((id) => inGroup.has(id));
    // No match skips the key: sanitizeSelectedTeamsForScope would widen it to ['all'].
    if (!kept.length) return { teams: null, reported: ['teams'] };
    return { teams: kept, reported: kept.length < requested.length ? ['teams'] : [] };

// Atoms matched case-insensitively against the option ids present now, each returned as the
// option's own id; an atom with no option is dropped.
function matchOptionIds(atoms, optionIds) {
    const byId = new Map((optionIds || []).map((id) => [normalizeId(id), id]));
    const matched = atoms.filter((atom) => byId.has(normalizeId(atom)));
    return { ids: unique(matched.map((atom) => byId.get(normalizeId(atom)))), dropped: matched.length < atoms.length };
}

// null = neutral, { only } or { hidden }. Returns { value } or null when the key is skipped.
function resolveMultiFacet(link, fallback, optionIds, reported) {
    if (link === undefined) return { value: fallback };
    if (link === null) return { value: null };
    const form = Array.isArray(link.only) ? 'only' : 'hidden';
    const { ids, dropped } = matchOptionIds(link[form], optionIds);
    if (dropped) reported.push('filters');
    return ids.length ? { value: { [form]: ids } } : null;
}

// undefined = neutral, [] = only Epics without a Project Track, else the listed option ids.
function resolveTrackFacet(link, optionIds, reported) {
    if (link === undefined) return { value: SHARE_VIEW_DEFAULTS.track };
    if (!link.length) return { value: [] };
    const { ids, dropped } = matchOptionIds(link, optionIds);
    if (dropped) reported.push('filters');
    return ids.length ? { value: ids } : null;
}

// options = { status, priority, track }: the facet option ids present when the scope has loaded.

// key is absent, so callers test Object.hasOwn(apply, key).
export function resolveShareViewFacets(state, options, mode = state.mode) {
    const apply = {};
    const reported = [];
    if (mode !== 'catch-up' && mode !== 'planning') return { apply, reported };
    // Priority is limited to the six canonical labels the facet is built from.
    const priorityIds = (options.priority || []).filter((id) => PRIORITY_AXIS.includes(id));
    const resolved = {
        status: resolveMultiFacet(state.status, SHARE_VIEW_DEFAULTS.status, options.status, reported),
        priority: resolveMultiFacet(state.priority, SHARE_VIEW_DEFAULTS.priority, priorityIds, reported),
        track: resolveTrackFacet(state.track, options.track, reported),
    };
    for (const [key, facet] of Object.entries(resolved)) {
        if (facet) apply[key] = facet.value;
    }
    return { apply, reported: unique(reported) };
}

    return { panel: value !== 'off', expanded: Object.fromEntries(keys.map((key) => [key, open.has(key)])) };
}

// The default (no strict scope) needs no readiness; a strict scope applies only once the Board is

}

}

function sprintRange(from, to, sprints) {
    const ordered = [...sprints].sort(compareSprintsChronologically);
    const start = ordered.findIndex((entry) => String(entry.id) === from);
    const end = ordered.findIndex((entry) => String(entry.id) === to);
    return start >= 0 && end >= start ? { from: String(ordered[start].id), to: String(ordered[end].id) } : null;
}

// Both bounds or neither, from <= to, quarters inside the 16-quarter window, Sprint ids in the
// catalog. Anything else falls back to the default range and reports `mode`.
function resolveStatsRange(state, view, scope, ctx, reported) {
    const quarterBased = view === 'cohort';
    if (!quarterBased && !SPRINT_RANGE_STATS_VIEWS.has(view)) return null;
    const currentQuarter = ctx.currentQuarter || getCurrentQuarterLabel();
    let fallback = { from: currentQuarter, to: currentQuarter };
    if (!quarterBased) {
        const { startSprintId, endSprintId } = buildDefaultExcludedCapacityRange(ctx.sprints, scope.sprint.id);
        fallback = { from: startSprintId, to: endSprintId };
    }
    const { from, to } = state;
    if (from === undefined && to === undefined) return fallback;
    // A lone bound matches nothing in either validator, so only complete pairs survive.
    const range = quarterBased ? quarterRange(from, to, currentQuarter) : sprintRange(from, to, ctx.sprints);
    if (!range) reported.push('mode');
    return range || fallback;
}

function focusApplies(candidate, state, mode) {
    if (candidate.kind === 'alertTeam') {
        return mode === 'catch-up' && state.alerts !== 'off' && ALERT_ANCHORS.has(candidate.anchor);
    }
    if (!FOCUS_KINDS.has(candidate.kind) || (mode !== 'catch-up' && mode !== 'planning')) return false;
    // The Planning table never switches its Epics/Stories rows: the target must be in the link's rows.
    if (mode === 'planning' && linkValue(state, 'layout') === 'table') {
        return candidate.kind === (linkValue(state, 'rows') === 'story' ? 'story' : 'epic');
    }
    return true;
}

// At most one target; alertTeam outranks epic and story. `dropped` is true when the link carried
// a target that was not chosen, so the notice names `focus`.
function selectFocus(state, mode) {
    const candidates = [];
    if (state.alertTeam) candidates.push({ kind: 'alertTeam', anchor: state.alertTeam.anchor, id: state.alertTeam.id });
    if (state.focus) candidates.push({ kind: state.focus.kind, key: state.focus.key });
    const target = candidates.find((candidate) => focusApplies(candidate, state, mode)) || null;
    return { target, dropped: candidates.length > (target ? 1 : 0) };
}

export function resolveShareViewFocus(state, mode) {
    return selectFocus(state, mode).target;
}

// Only the keys that apply to the resolved mode, each the link value or its default constant.
// Teams, facets and focus are resolved by their own functions once their data has loaded.
export function buildShareViewFieldPlan(state, scope, ctx = {}) {
    const apply = {};
    const reported = [];
    const take = (...keys) => keys.forEach((key) => { apply[key] = linkValue(state, key); });
    if (scope.mode === 'board') {
        take('find');
        const boardScope = resolveBoardScope(state, scope, ctx, reported);
        if (boardScope !== undefined) apply.scope = boardScope;
    } else if (scope.mode === 'statistics') {
        take('stats');
        apply.range = resolveStatsRange(state, apply.stats, scope, ctx, reported);
    } else {
        take('find', 'lanes');
        if (scope.mode === 'planning') {
            take('layout');
            if (apply.layout === 'table') take('rows');
            else take('initiatives', 'sort');
        } else {
            take('initiatives', 'sort');
            apply.alerts = resolveAlerts(state.alerts);
        }
    }
    if (selectFocus(state, scope.mode).dropped) reported.push('focus');
    return { apply, reported: unique(reported) };
}

```

```
- Semantics Tasks 8, 9 and 11 rely on:
  - `initShareViewFromLocation` captures `location.search`, strips the schema keys with `history.replaceState(history.state, '', pathname + remainingSearch + hash)` (only when a key was present), then decodes the captured string. `none` leaves the store `absent`; `ok` makes it `pending` with `entry = { state, reported }`; `newer` and `invalid` make it `dropped` at once with a `newer` or `invalid` notice. It never throws: an unreadable location means no link, a `SecurityError` leaves the link in the address bar but still applies it, and a throwing decoder drops the link as `invalid`.
  - `getShareViewSnapshot()` returns the same frozen object until something changes; `subscribeShareView` and `getShareViewSnapshot` are the two functions `useSyncExternalStore` takes.
  - `hold` is true while the phase is `pending` or `running` and `markShareViewScopeCommitted()` has not run. It is false in every other phase, so a terminal state always releases the ENG task load. A same-scope link calls `markShareViewScopeCommitted()` without ever running Stage A.
  - `finishShareView` ends an active apply once: it returns `true` for that call and `false` for any later call, an unknown result, or a store that is not `pending` or `running`, and it changes nothing in those cases (so a caller emits its single analytics event only on `true`). `partial` and `dropped` set `notice = { kind: result, categories }` with duplicates removed; `done` and `cancelled` set no notice. `stage` keeps its last value after the end.
  - The Planning row seed is one-shot module state outside the snapshot. It is not cleared when the apply ends, so a table that mounts after a cancel still takes the Stage B value.
  - Nothing here touches browser storage or the network.

// One module-scope store for a shared-link apply. entry.js fills it before React mounts, the apply
// hook drives it through the stages, and useSyncExternalStore reads it. Nothing here persists: a
// reload reads the already stripped URL, finds no link and applies nothing.
import { decodeShareView, stripShareViewParams } from './codec.js';

export const SHARE_VIEW_PHASES = Object.freeze({
    ABSENT: 'absent',
    PENDING: 'pending',
    RUNNING: 'running',
    DONE: 'done',
    PARTIAL: 'partial',
    DROPPED: 'dropped',
    CANCELLED: 'cancelled',
});
const { ABSENT, PENDING, RUNNING, DONE, PARTIAL, DROPPED, CANCELLED } = SHARE_VIEW_PHASES;
const TERMINAL = new Set([DONE, PARTIAL, DROPPED, CANCELLED]);
// Cancelled applies stay silent; done has nothing to report.
const NOTICE_RESULTS = new Set([PARTIAL, DROPPED]);
const ROW_MODES = new Set(['epic', 'story']);

const listeners = new Set();
let scopeCommitted = false;
let planningRowMode = null;

const isActive = (phase) => phase === PENDING || phase === RUNNING;

// hold keeps the ENG task load from fetching the recipient's saved scope while a link is pending
// or running and its own scope has not been committed yet.
function createSnapshot(fields) {
    return Object.freeze({ ...fields, hold: isActive(fields.phase) && !scopeCommitted, terminal: TERMINAL.has(fields.phase) });
}
const emptySnapshot = () => createSnapshot({ phase: ABSENT, stage: null, entry: null, notice: null });
let snapshot = emptySnapshot();

function commit(patch) {
    snapshot = createSnapshot({ ...snapshot, ...patch });
    listeners.forEach((listener) => listener());
}

export const getShareViewSnapshot = () => snapshot;

export function subscribeShareView(listener) {
    listeners.add(listener);
    return () => { listeners.delete(listener); };
}

export function beginShareViewStage(stage) {
    if (!isActive(snapshot.phase) || (snapshot.phase === RUNNING && snapshot.stage === stage)) return;
    commit({ phase: RUNNING, stage });
}

export function markShareViewScopeCommitted() {
    if (scopeCommitted || !isActive(snapshot.phase)) return;
    scopeCommitted = true;
    commit({});
}

// True only for the call that ended the apply, so the caller reports it exactly once.
export function finishShareView(result, categories = []) {
    if (!TERMINAL.has(result) || !isActive(snapshot.phase)) return false;
    commit({
        phase: result,
        notice: NOTICE_RESULTS.has(result) ? { kind: result, categories: [...new Set(categories)] } : null,
    });
    return true;
}

export function dismissShareViewNotice() {
    if (snapshot.notice) commit({ notice: null });
}

// The Planning table's Epics/Stories mode is local state that resets on every mount, so a link
// seeds the next mount once instead of owning that state.
export function setPendingPlanningRowMode(mode) {
    planningRowMode = ROW_MODES.has(mode) ? mode : null;
}

export function takeInitialPlanningRowMode() {
    const mode = planningRowMode;
    planningRowMode = null;
    return mode;
}

export function resetShareViewForTests() {
    listeners.clear();
    scopeCommitted = false;
    planningRowMode = null;
    snapshot = emptySnapshot();
}

function readSearch(win) {
    try {
        return String(win.location.search || '');
    } catch {
        return '';
    }
}

// Removes the schema keys from the address bar and keeps every other param and the hash. A
// SecurityError or a codec fault leaves the URL as it was; there is nothing safer to fall back to.
function stripSchemaKeys(win, search) {
    try {
        const remaining = stripShareViewParams(search);
        if (remaining === search) return;
        win.history.replaceState(win.history.state, '', `${win.location.pathname}${remaining}${win.location.hash}`);
    } catch {
        // The link stays in the address bar.
    }
}

export function initShareViewFromLocation(win = window) {
    const search = readSearch(win);
    // Strip first: nothing that reads the store or the URL afterwards can see the link.
    stripSchemaKeys(win, search);
    let decoded;
    try {
        decoded = decodeShareView(search);
    } catch {
        decoded = { kind: 'invalid' };
    }
    if (decoded.kind === 'ok') {
        commit({ phase: PENDING, entry: { state: decoded.state, reported: [...decoded.reported] } });
    } else if (decoded.kind === 'newer' || decoded.kind === 'invalid') {
        commit({ phase: DROPPED, notice: { kind: decoded.kind, categories: [] } });
    }
}
